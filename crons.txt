5. Accept N Numbers and Print Average
#!/bin/bash
echo "Enter how many numbers:"
read n
sum=0
for ((i=1; i<=n; i++))
do
    echo "Enter number $i:"
    read num
    sum=$((sum + num))
done

average=$((sum / n))
echo "Sum = $sum"
echo "Average = $average"

12. Display Files Sorted by Size
#!/bin/bash
echo "Enter directory:"
read dir
find "$dir" -type f -printf '%s %p\n' | sort -nr | while read size file
do
    filename=$(basename "$file")
    owner=$(stat -c '%U' "$file")
    date=$(stat -c '%y' "$file")

    echo "Filename: $filename"
    echo "Size: $size bytes"
    echo "Owner: $owner"
    echo "Last Modified: $date"
    echo "-------------------------"
done

13. Interactive System Administration Menu
#!/bin/bash
while true
do
    echo "---------- MENU ----------"
    echo "1. Display disk usage"
    echo "2. Display top 10 CPU processes"
    echo "3. Show logged-in users"
    echo "4. Backup home directory"
    echo "5. Exit"
    echo "Enter choice:"
    read choice
    case $choice in

        1)
            df -h
            ;;

        2)
            ps -eo pid,user,%cpu,comm --sort=-%cpu | head -11
            ;;

        3)
            who
            ;;

        4)
            tar -czf "home-backup-$(date +%Y-%m-%d).tar.gz" "$HOME"
            echo "Home directory backup created."
            ;;

        5)
            echo "Exiting..."
            break
            ;;

        *)
            echo "Invalid choice"
            ;;
    esac
done

19. Accept Unlimited Numbers as Parameters
#!/bin/bash
if [ $# -eq 0 ]
then
    echo "Please provide numbers as parameters."
    exit 1
fi
count=$#
sum=0
highest=$1
lowest=$1
for num in "$@"
do
    sum=$((sum + num))

    if [ $num -gt $highest ]
    then
        highest=$num
    fi

    if [ $num -lt $lowest ]
    then
        lowest=$num
    fi
done
average=$((sum / count))
echo "Total count = $count"
echo "Sum = $sum"
echo "Average = $average"
echo "Highest = $highest"
echo "Lowest = $lowest"
